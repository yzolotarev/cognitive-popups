def build_probe_prompt(source: str, *, existing: list[Mapping[str, Any]] | None = None,
                       invariant: str = "", type_id: int | None = None) -> str:

        "properties": {
            "type": {"type": "integer", "enum": [type_id] if type_id is not None else [0, 1, 2, 3]},
            "title": {"type": "string"}, "why": {"type": "string"},
            "invariant": {"type": "string", **({"const": invariant} if invariant else {})},

        },
    }

        "Каждая проба — ОДНА маленькая операция над конкретным материалом: "
